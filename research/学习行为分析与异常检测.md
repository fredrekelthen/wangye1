# 学习行为分析与异常检测系统设计方案

## 一、研究背景与意义

学习行为分析是智能教育系统的重要组成部分，通过对学习者的学习行为数据进行采集、特征提取和模式分析，可以实现学习状态识别、异常行为检测和智能干预。

### 1.1 研究价值

- **理解学习过程**：深入了解学习者的学习模式和习惯
- **异常检测**：及时发现学习困难、倦怠等异常情况
- **个性化干预**：基于分析结果提供针对性支持
- **教学优化**：为教师提供决策支持

### 1.2 研究目标

本研究旨在设计并实现一个完整的学习行为分析与异常检测系统：

1. 多维度行为数据采集
2. 行为特征提取与分析
3. 学习状态识别与预测
4. 异常行为检测与预警
5. 智能干预建议生成

"""
时序特征提取器
从学习行为时序数据中提取特征
"""
import numpy as np
from scipy import stats
from collections import defaultdict

class TemporalFeatureExtractor:
    """
    时序特征提取器
    """
    
    def extract_features(self, events, time_window='1H'):
        """
        提取时序特征
        
        events: List[LearningEvent] - 学习事件列表
        time_window: str - 时间窗口大小
        """
        features = {}
        
        # 按时间窗口分组
        windows = self.group_by_time_window(events, time_window)
        
        # 1. 活动强度特征
        features['event_count'] = len(events)
        features['events_per_hour'] = len(events) / max(1, self.get_duration_hours(events))
        
        # 2. 活跃时段特征
        features['morning_activity'] = self.calculate_time_period_activity(events, 'morning')
        features['afternoon_activity'] = self.calculate_time_period_activity(events, 'afternoon')
        features['evening_activity'] = self.calculate_time_period_activity(events, 'evening')
        features['night_activity'] = self.calculate_time_period_activity(events, 'night')
        
        # 3. 时间分布特征
        features['session_count'] = len(self.get_sessions(events))
        features['avg_session_duration'] = self.calculate_avg_session_duration(events)
        features['avg_session_gap'] = self.calculate_avg_session_gap(events)
        
        # 4. 周期性特征
        features['is_weekend_learner'] = self.is_weekend_learner(events)
        features['study_time_variance'] = self.calculate_variance_by_day_of_week(events)
        
        # 5. 稳定性特征
        features['event_rate_cv'] = self.calculate_coefficient_of_variation(
            self.get_event_rates_per_day(events)
        )
        
        return features
    
    def calculate_time_period_activity(self, events, period):
        """计算特定时段的活跃度"""
        if period == 'morning':
            hours = range(6, 12)
        elif period == 'afternoon':
            hours = range(12, 18)
        elif period == 'evening':
            hours = range(18, 22)
        else:  # night
            hours = [22, 23, 0, 1, 2, 3, 4, 5]
        
        count = sum(1 for e in events if e.timestamp.hour in hours)
        return count
    
    def get_sessions(self, events):
        """识别学习会话"""
        sessions = []
        current_session = []
        gap_threshold = 30 * 60  # 30分钟无活动视为会话结束
        
        sorted_events = sorted(events, key=lambda e: e.timestamp)
        
        for event in sorted_events:
            if not current_session:
                current_session.append(event)
            else:
                time_gap = (event.timestamp - current_session[-1].timestamp).seconds
                if time_gap > gap_threshold:
                    sessions.append(current_session)
                    current_session = [event]
                else:
                    current_session.append(event)
        
        if current_session:
            sessions.append(current_session)
        
        return sessions
    
    def calculate_avg_session_duration(self, events):
        """计算平均会话时长"""
        sessions = self.get_sessions(events)
        if not sessions:
            return 0
        
        durations = []
        for session in sessions:
            if len(session) > 1:
                duration = (session[-1].timestamp - session[0].timestamp).seconds / 60
                durations.append(duration)
        
        return np.mean(durations) if durations else 0

        features['question_attempt_count'] = len(answer_events)
        features['question_correct_rate'] = self.calculate_correct_rate(answer_events)
        features['avg_answer_time'] = self.calculate_avg_answer_time(answer_events)
        features['retry_rate'] = self.calculate_retry_rate(answer_events)
        
        # 4. 资源交互特征
        features['note_count'] = self.count_event_type(events, 'NOTE_CREATE')
        features['bookmark_count'] = self.count_event_type(events, 'BOOKMARK_ADD')
        features['download_count'] = self.count_event_type(events, 'DOWNLOAD_RESOURCE')
        features['search_count'] = self.count_event_type(events, 'SEARCH_QUERY')
        
        # 5. 社交交互特征
        features['discussion_count'] = self.count_event_type(events, 'DISCUSSION_POST')
        features['comment_count'] = self.count_event_type(events, 'COMMENT_CREATE')
        
        return features
    
    def calculate_video_completion_rate(self, video_events):
        """计算视频完成率"""
        completes = self.filter_by_type(video_events, 'VIDEO_COMPLETE')
        plays = self.filter_by_type(video_events, 'VIDEO_PLAY')
        
        if not plays:
            return 0
        
        return len(completes) / len(plays)
    
    def calculate_retry_rate(self, answer_events):
        """计算题目重做率"""
        retries = self.filter_by_type(answer_events, 'QUESTION_RETRY')
        attempts = self.filter_by_type(answer_events, 'QUESTION_ANSWER')
        
        if not attempts:
            return 0
        
        return len(retries) / (len(attempts) + len(retries))

public class CognitiveFeatureExtractor {

     * 提取认知特征
     */
    public CognitiveFeatures extractCognitiveFeatures(

        features.setUnderstandingLevel(assessUnderstanding(events));
        
        // 3. 学习策略特征
        features.setStrategyType(classifyLearningStrategy(events));
        features.setDeepLearningRatio(calculateDeepLearningRatio(events));
        features.setSurfaceLearningRatio(calculateSurfaceLearningRatio(events));
        
        // 4. 元认知特征
        features.setSelfRegulationScore(assessSelfRegulation(events));
        features.setPlanningScore(assessPlanning(events));
        features.setMonitoringScore(assessMonitoring(events));
        
        return features;
    }
    
    /**
     * 估计认知负荷
     */
    private double estimateCognitiveLoad(List<LearningEvent> events) {
        double load = 0.5;  // 基础负荷
        
        // 视频播放速度慢 → 高负荷
        double avgPlaySpeed = calculateAvgPlaySpeed(events);
        if (avgPlaySpeed < 0.8) {
            load += 0.2;
        }
        
        // 频繁暂停 → 高负荷
        double pauseRate = calculatePauseRate(events);
        if (pauseRate > 0.3) {
            load += 0.15;
        }
        
        // 视频重复观看 → 高负荷
        double replayRate = calculateReplayRate(events);
        if (replayRate > 0.2) {
            load += 0.1;
        }
        
        // 错误率高 → 高负荷
        double errorRate = calculateErrorRate(events);
        if (errorRate > 0.5) {
            load += 0.15;
        }
        
        // 答题时间长 → 高负荷（对于难题）
        double avgAnswerTime = calculateAvgAnswerTime(events);
        if (avgAnswerTime > 180) {  // 超过3分钟
            load += 0.1;
        }
        
        return Math.min(1.0, load);
    }
    
    /**
     * 检测困惑指标
     */
    private List<ConfusionIndicator> detectConfusion(List<LearningEvent> events) {
        List<ConfusionIndicator> indicators = new ArrayList<>();
        
        // 1. 频繁回看
        int backwatchCount = countBackwatch(events);
        if (backwatchCount > 3) {
            ConfusionIndicator indicator = new ConfusionIndicator();
            indicator.setType(ConfusionType.FREQUENT_BACKWATCH);
            indicator.setSeverity(backwatchCount > 5 ? Severity.HIGH : Severity.MEDIUM);
            indicator.setDescription("频繁回看视频，可能存在理解困难");
            indicators.add(indicator);
        }
        
        // 2. 犹豫行为
        int hesitationCount = countHesitation(events);
        if (hesitationCount > 5) {
            ConfusionIndicator indicator = new ConfusionIndicator();
            indicator.setType(ConfusionType.HESITATION);
            indicator.setSeverity(Severity.MEDIUM);
            indicator.setDescription("答题犹豫时间较长");
            indicators.add(indicator);
        }
        
        // 3. 重复浏览
        int revisitCount = countPageRevisit(events);
        if (revisitCount > 3) {
            ConfusionIndicator indicator = new ConfusionIndicator();
            indicator.setType(ConfusionType.REPETITIVE_REVIEW);
            indicator.setSeverity(Severity.LOW);
            indicator.setDescription("重复浏览同一页面");
            indicators.add(indicator);
        }
        
        // 4. 错误后立即重试
        int immediateRetryCount = countImmediateRetry(events);
        if (immediateRetryCount > 2) {
            ConfusionIndicator indicator = new ConfusionIndicator();
            indicator.setType(ConfusionType.IMMEDIATE_RETRY);
            indicator.setSeverity(Severity.MEDIUM);
            indicator.setDescription("错误后立即重试，可能未理解正确答案");
            indicators.add(indicator);
        }
        
        return indicators;
    }
    
    /**
     * 分类学习策略
     */
    private LearningStrategy classifyLearningStrategy(List<LearningEvent> events) {
        double deepScore = calculateDeepLearningRatio(events);
        double surfaceScore = calculateSurfaceLearningRatio(events);
        
        if (deepScore > 0.6 && surfaceScore < 0.3) {
            return LearningStrategy.DEEP_LEARNING;
        } else if (surfaceScore > 0.6 && deepScore < 0.3) {
            return LearningStrategy.SURFACE_LEARNING;
        } else if (deepScore > 0.4 && surfaceScore > 0.4) {
            return LearningStrategy.STRATEGIC_LEARNING;
        } else {
            return LearningStrategy.MIXED_LEARNING;
        }
    }
}

        LearningFeatures features = extractFeatures(userId, courseId);

    }
    
    /**
     * 识别专注度
     */
    public FocusLevel recognizeFocusLevel(List<LearningEvent> events) {
        // 计算专注度指标
        double attentionScore = calculateAttentionScore(events);
        double engagementScore = calculateEngagementScore(events);
        double consistencyScore = calculateConsistencyScore(events);
        
        // 综合评分
        double overallScore = 
            attentionScore * 0.4 + 
            engagementScore * 0.3 + 
            consistencyScore * 0.3;
        
        // 分类
        if (overallScore >= 0.8) {
            return FocusLevel.HIGH;
        } else if (overallScore >= 0.6) {

                                                 List<LearningEvent> events) {

        double frustration = calculateFrustrationLevel(events);
        double engagement = calculateEngagementScore(events);
        double satisfaction = calculateSatisfactionLevel(events);
        
        EmotionalState state = new EmotionalState();
        state.setFrustrationLevel(frustration);
        state.setEngagementLevel(engagement);
        state.setSatisfactionLevel(satisfaction);
        
        // 综合判断
        if (frustration > 0.6 && engagement < 0.3) {
            state.setType(EmotionalType.FRUSTRATED);
        } else if (engagement > 0.7 && satisfaction > 0.6) {
            state.setType(EmotionalType.ENGAGED);
        } else if (satisfaction < 0.3 && engagement < 0.3) {
            state.setType(EmotionalType.BORED);
        } else if (engagement > 0.5 && frustration < 0.4) {
            state.setType(EmotionalType.FLOW);
        } else {
            state.setType(EmotionalType.NEUTRAL);
        }
        
        return state;
    }
    
    /**
     * 计算注意力得分
     */
    private double calculateAttentionScore(List<LearningEvent> events) {
        double score = 0.5;
        
        // 页面停留时间合理
        double avgDuration = calculateAvgPageDuration(events);
        if (avgDuration >= 30 && avgDuration <= 600) {  // 30秒到10分钟
            score += 0.2;
        }
        
        // 视频观看完整
        double videoCompletionRate = calculateVideoCompletionRate(events);
        score += videoCompletionRate * 0.2;
        
        // 交互频率适中
        double interactionRate = calculateInteractionRate(events);
        if (interactionRate >= 0.1 && interactionRate <= 2.0) {
            score += 0.1;
        }
        
        return Math.min(1.0, score);
    }
    
    /**
     * 计算挫败感水平
     */
    private double calculateFrustrationLevel(List<LearningEvent> events) {
        double level = 0;
        
        // 高错误率
        double errorRate = calculateErrorRate(events);
        level += errorRate * 0.3;
        
        // 频繁重试
        double retryRate = calculateRetryRate(events);
        level += retryRate * 0.2;
        
        // 长时间无进展
        double noProgressTime = calculateNoProgressTime(events);
        if (noProgressTime > 600) {  // 超过10分钟
            level += 0.2;
        }
        
        // 低完成率
        double completionRate = calculateTaskCompletionRate(events);
        if (completionRate < 0.3) {
            level += 0.2;
        }
        
        return Math.min(1.0, level);
    }

"""
学习状态预测模型
使用LSTM进行时序预测
"""
import torch
import torch.nn as nn
from torch.utils.data import Dataset, DataLoader

class LearningStatePredictor(nn.Module):
    """
    基于LSTM的学习状态预测模型
    预测学习者未来的学习状态和表现
    """
    
    def __init__(self, input_dim, hidden_dim, num_layers, output_dim):
        super().__init__()
        
        self.hidden_dim = hidden_dim
        self.num_layers = num_layers
        
        # LSTM层
        self.lstm = nn.LSTM(
            input_dim, 
            hidden_dim, 
            num_layers, 
            batch_first=True,
            dropout=0.3
        )
        
        # 全连接层
        self.fc1 = nn.Linear(hidden_dim, hidden_dim // 2)
        self.fc2 = nn.Linear(hidden_dim // 2, output_dim)
        
        # 输出激活
        self.sigmoid = nn.Sigmoid()
        self.softmax = nn.Softmax(dim=-1)
        
    def forward(self, x):
        """
        x: (batch_size, seq_len, input_dim)
        """
        # LSTM前向传播
        lstm_out, (hidden, cell) = self.lstm(x)
        
        # 取最后一个时间步的输出
        last_output = lstm_out[:, -1, :]
        
        # 全连接层
        out = self.fc1(last_output)
        out = torch.relu(out)
        out = self.fc2(out)
        
        return out
    
    def predict_next_state(self, historical_features, prediction_horizon=7):
        """
        预测未来学习状态
        
        historical_features: 历史特征序列 (seq_len, input_dim)
        prediction_horizon: 预测天数
        """
        self.eval()
        
        with torch.no_grad():
            # 转换为tensor
            x = torch.FloatTensor(historical_features).unsqueeze(0)
            
            # 预测
            predictions = []
            current_seq = x
            
            for _ in range(prediction_horizon):
                pred = self.forward(current_seq)
                predictions.append(pred.numpy()[0])
                
                # 更新序列
                current_seq = torch.cat([
                    current_seq[:, 1:, :], 
                    pred.unsqueeze(1)
                ], dim=1)
            
            return np.array(predictions)


class StatePredictionDataset(Dataset):
    """预测数据集"""
    
    def __init__(self, features, labels, seq_length=14):
        self.features = features
        self.labels = labels
        self.seq_length = seq_length
        
    def __len__(self):
        return len(self.features) - self.seq_length
    
    def __getitem__(self, idx):
        X = self.features[idx:idx + self.seq_length]
        y = self.labels[idx + self.seq_length]
        return torch.FloatTensor(X), torch.FloatTensor(y)


def train_prediction_model(train_data, val_data, epochs=100):
    """训练预测模型"""
    
    # 数据准备
    train_dataset = StatePredictionDataset(
        train_data['features'], 
        train_data['labels']
    )
    val_dataset = StatePredictionDataset(
        val_data['features'], 
        val_data['labels']
    )
    
    train_loader = DataLoader(train_dataset, batch_size=32, shuffle=True)
    val_loader = DataLoader(val_dataset, batch_size=32)
    
    # 模型初始化
    model = LearningStatePredictor(
        input_dim=50,  # 特征维度
        hidden_dim=128,
        num_layers=2,
        output_dim=10  # 预测维度
    )
    
    optimizer = torch.optim.Adam(model.parameters(), lr=0.001)
    criterion = nn.MSELoss()
    
    # 训练循环
    best_val_loss = float('inf')
    
    for epoch in range(epochs):
        # 训练
        model.train()
        train_loss = 0
        for X, y in train_loader:
            optimizer.zero_grad()
            output = model(X)
            loss = criterion(output, y)
            loss.backward()
            optimizer.step()
            train_loss += loss.item()
        
        # 验证
        model.eval()
        val_loss = 0
        with torch.no_grad():
            for X, y in val_loader:
                output = model(X)
                loss = criterion(output, y)
                val_loss += loss.item()
        
        train_loss /= len(train_loader)
        val_loss /= len(val_loader)
        
        # 保存最佳模型
        if val_loss < best_val_loss:
            best_val_loss = val_loss
            torch.save(model.state_dict(), 'best_model.pth')
        
        if epoch % 10 == 0:
            print(f"Epoch {epoch}, Train Loss: {train_loss:.4f}, Val Loss: {val_loss:.4f}")
    
    return model

        );

## 九、实验设计与评估

### 9.1 评估指标

| 指标类别 | 指标名称 | 描述 | 计算方法 |
|---------|---------|------|---------|
| **检测性能** | 准确率 | 正确检测异常的比例 | TP/(TP+FP) |
| | 召回率 | 异常被检测出的比例 | TP/(TP+FN) |
| | F1分数 | 准确率和召回率的调和平均 | 2*P*R/(P+R) |
| **预测性能** | MAE | 平均绝对误差 | 平均|预测-实际| |
| | RMSE | 均方根误差 | sqrt(平均(预测-实际)²) |
| **用户价值** | 预警有效性 | 预警后行为改善比例 | 改善数/预警数 |
| | 干预接受率 | 用户接受干预建议的比例 | 接受数/建议数 |
