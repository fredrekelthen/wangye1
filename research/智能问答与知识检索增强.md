# 智能问答与知识检索增强系统设计方案

## 一、研究背景与意义

智能问答系统是教育智能体的核心交互能力之一。通过构建增强型检索系统，结合知识图谱和多模态检索技术，可以显著提升问答系统的准确性和用户体验。

### 1.1 研究价值

- **精准回答**：结合RAG技术，提升回答准确性
- **知识溯源**：引用来源，增强可信度
- **上下文理解**：支持多轮对话，保持上下文连贯
- **知识关联**：通过知识图谱发现知识点间的关联

### 1.2 研究目标

本研究旨在设计并实现一个增强型智能问答与知识检索系统：

1. 多粒度检索能力（文档、段落、知识点）
2. 混合检索策略（向量+关键词+知识图谱）
3. 上下文感知的答案生成
4. 可解释的引用溯源机制

```java
/**
 * 问题理解服务
 * 解析用户问题，提取关键信息
 */
@Service
public class QuestionUnderstandingService {
    
    private final NERService nerService;
    private final IntentClassifier intentClassifier;
    private final KeywordExtractor keywordExtractor;
    
    /**
     * 理解问题
     */
    public QuestionAnalysis analyze(String question) {
        QuestionAnalysis analysis = new QuestionAnalysis();
        
        // 1. 基础处理
        String normalized = normalize(question);
        analysis.setOriginalQuestion(question);
        analysis.setNormalizedQuestion(normalized);
        
        // 2. 意图识别
        Intent intent = intentClassifier.classify(normalized);
        analysis.setIntent(intent);
        
        // 3. 实体抽取
        List<Entity> entities = nerService.extract(normalized);
        analysis.setEntities(entities);
        
        // 4. 关键词提取
        List<String> keywords = keywordExtractor.extract(normalized);
        analysis.setKeywords(keywords);
        
        // 5. 问题类型分类
        QuestionType type = classifyQuestionType(normalized);
        analysis.setQuestionType(type);
        
        // 6. 查询构建
        String query = buildQuery(analysis);
        analysis.setSearchQuery(query);
        
        return analysis;
    }
    
    /**
     * 标准化问题
     */
    private String normalize(String question) {
        // 去除多余空格
        question = question.trim().replaceAll("\\s+", " ");
        
        // 统一标点
        question = question.replace("?", "？");
        
        return question;
    }
    
    /**
     * 分类问题类型
     */
    private QuestionType classifyQuestionType(String question) {
        if (question.matches(".*什么是.*|.*定义.*")) {
            return QuestionType.DEFINITION;
        }
        if (question.matches(".*为什么.*")) {
            return QuestionType.REASON;
        }
        if (question.matches(".*如何.*|.*怎么做.*")) {
            return QuestionType.PROCEDURAL;
        }
        if (question.matches(".*比较.*|.*区别.*")) {
            return QuestionType.COMPARISON;
        }
        if (question.matches(".*例子.*|.*示例.*")) {
            return QuestionType.EXAMPLE;
        }
        if (question.matches(".*步骤.*|.*流程.*")) {
            return QuestionType.PROCESS;
        }
        return QuestionType.GENERAL;
    }
    
    /**
     * 构建检索查询
     */
    private String buildQuery(QuestionAnalysis analysis) {
        StringBuilder query = new StringBuilder();
        
        // 添加关键词
        for (String keyword : analysis.getKeywords()) {
            query.append(keyword).append(" ");
        }
        
        // 添加实体
        for (Entity entity : analysis.getEntities()) {
            query.append(entity.getText()).append(" ");
        }
        
        // 添加意图修饰
        if (analysis.getIntent() == Intent.EXPLANATION) {
            query.append("解释 说明 ");
        } else if (analysis.getIntent() == Intent.DEFINITION) {
            query.append("定义 概念 ");
        }
        
        return query.toString().trim();
    }
}

@Service
public class IntentClassifier {
    
    private final Map<Intent, Pattern> intentPatterns;
    
    @PostConstruct
    public void init() {
        intentPatterns = new HashMap<>();
        intentPatterns.put(Intent.DEFINITION, 
            Pattern.compile("什么是|定义|概念"));
        intentPatterns.put(Intent.EXPLANATION, 
            Pattern.compile("为什么|原因|解释"));
        intentPatterns.put(Intent.PROCEDURAL, 
            Pattern.compile("如何|怎么|怎样|步骤"));
        intentPatterns.put(Intent.COMPARISON, 
            Pattern.compile("比较|区别|不同"));
        intentPatterns.put(Intent.EXAMPLE, 
            Pattern.compile("例子|示例|举例"));
        intentPatterns.put(Intent.FACTUAL, 
            Pattern.compile("是谁|在哪|什么时候"));
    }
    
    public Intent classify(String question) {
        for (Map.Entry<Intent, Pattern> entry : intentPatterns.entrySet()) {
            if (entry.getValue().matcher(question).find()) {
                return entry.getKey();
            }
        }
        return Intent.GENERAL;
    }
}

@Service
public class NERService {
    
    /**
     * 抽取实体
     */
    public List<Entity> extract(String text) {
        List<Entity> entities = new ArrayList<>();
        
        // 使用规则抽取
        entities.addAll(extractConcepts(text));
        entities.addAll(extractPeople(text));
        entities.addAll(extractTechnicalTerms(text));
        
        // 去重
        return entities.stream()
            .distinct()
            .collect(Collectors.toList());
    }
    
    /**
     * 抽取概念实体
     */
    private List<Entity> extractConcepts(String text) {
        List<Entity> concepts = new ArrayList<>();
        
        // 匹配概念模式：XXX是一种、XXX定义为、XXX的概念
        Pattern pattern = Pattern.compile(
            "([\\u4e00-\\u9fa5a-zA-Z0-9]+)是一种|" +
            "([\\u4e00-\\u9fa5a-zA-Z0-9]+)定义为|" +
            "([\\u4e00-\\u9fa5a-zA-Z0-9]+)的概念"
        );
        
        Matcher matcher = pattern.matcher(text);
        while (matcher.find()) {
            for (int i = 1; i <= matcher.groupCount(); i++) {
                if (matcher.group(i) != null) {
                    Entity entity = new Entity();
                    entity.setText(matcher.group(i));
                    entity.setType(EntityType.CONCEPT);
                    concepts.add(entity);
                }
            }
        }
        
        return concepts;
    }
}
```

```java
@Service
public class VectorSearchService {
    
    private final EmbeddingModel embeddingModel;
    private final MilvusClient milvusClient;
    private final VectorIndexService indexService;
    
    /**
     * 语义检索
     */
    public List<SearchResult> semanticSearch(String query, int topK) {
        // 1. 生成查询向量
        double[] queryVector = embeddingModel.embed(query);
        
        // 2. 执行向量检索
        SearchParams searchParams = SearchParams.builder()
            .collectionName("knowledge_vectors")
            .vector(queryVector)
            .topK(topK)
            .metricType(MetricType.COSINE)
            .build();
        
        SearchResults results = milvusClient.search(searchParams);
        
        // 3. 处理结果
        return results.getResults().stream()
            .map(this::convertToSearchResult)
            .collect(Collectors.toList());
    }
    
    /**
     * 混合向量检索（结合关键词）
     */
    public List<SearchResult> hybridVectorSearch(
            String query, 
            List<String> keywords,
            int topK) {
        
        // 1. 查询向量
        double[] queryVector = embeddingModel.embed(query);
        
        // 2. 生成关键词增强查询
        String enhancedQuery = query + " " + String.join(" ", keywords);
        double[] enhancedVector = embeddingModel.embed(enhancedQuery);
        
        // 3. 加权融合查询向量
        double[] fusedVector = new double[queryVector.length];
        for (int i = 0; i < queryVector.length; i++) {
            fusedVector[i] = 0.7 * queryVector[i] + 
                            0.3 * enhancedVector[i];
        }
        
        // 4. 执行检索
        SearchParams params = SearchParams.builder()
            .collectionName("knowledge_vectors")
            .vector(fusedVector)
            .topK(topK)
            .build();
        
        return milvusClient.search(params)
            .getResults().stream()
            .map(this::convertToSearchResult)
            .collect(Collectors.toList());
    }
    
    /**
     * 向量索引构建
     */
    public void buildIndex(List<DocumentChunk> chunks) {
        List<EmbeddingResult> embeddings = new ArrayList<>();
        
        for (DocumentChunk chunk : chunks) {
            double[] vector = embeddingModel.embed(chunk.getContent());
            
            EmbeddingResult embedding = new EmbeddingResult();
            embedding.setId(chunk.getId());
            embedding.setVector(vector);
            embedding.setText(chunk.getContent());
            embedding.setMetadata(chunk.getMetadata());
            
            embeddings.add(embedding);
        }
        
        // 批量插入
        indexService.insert(embeddings);
        
        // 构建索引
        indexService.buildIndex();
    }
}

@Service
public class EmbeddingModel {
    
    private final OkHttpClient httpClient;
    private final String apiEndpoint;
    private final String apiKey;
    
    /**
     * 生成文本嵌入
     */
    public double[] embed(String text) {
        try {
            // 调用GLM API生成嵌入
            RequestBody body = RequestBody.create(
                MediaType.parse("application/json"),
                String.format("""
                    {
                        "input": "%s",
                        "model": "embedding-2"
                    }
                    """, text.replace("\"", "\\\""))
            );
            
            Request request = new Request.Builder()
                .url(apiEndpoint)
                .addHeader("Authorization", "Bearer " + apiKey)
                .post(body)
                .build();
            
            try (Response response = httpClient.newCall(request).execute()) {
                String responseBody = response.body().string();
                JSONObject json = new JSONObject(responseBody);
                
                // 解析嵌入向量
                JSONArray data = json.getJSONArray("data");
                JSONArray embedding = data.getJSONObject(0)
                    .getJSONArray("embedding");
                
                double[] vector = new double[embedding.length()];
                for (int i = 0; i < embedding.length(); i++) {
                    vector[i] = embedding.getDouble(i);
                }
                
                return vector;
            }
        } catch (Exception e) {
            throw new RuntimeException("Failed to generate embedding", e);
        }
    }
}
```

### 4.2 BM25关键词检索

```java
/**
 * BM25检索服务
 * 基于传统关键词的检索方法
 */
@Service
public class BM25SearchService {
    
    private final Map<String, Map<String, Integer>> invertedIndex;
    private final Map<String, Double> documentLengths;
    private final double avgDocumentLength;
    private final double k1 = 1.5;  // BM25参数
    private final double b = 0.75;  // BM25参数
    
    /**
     * 构建倒排索引
     */
    public void buildIndex(List<DocumentChunk> documents) {
        invertedIndex = new HashMap<>();
        documentLengths = new HashMap<>();
        
        int totalLength = 0;
        
        for (DocumentChunk doc : documents) {
            String docId = doc.getId();
            String content = doc.getContent();
            
            // 分词
            List<String> terms = tokenize(content);
            documentLengths.put(docId, (double) terms.size());
            totalLength += terms.size();
            
            // 构建倒排索引
            Map<String, Integer> termFreq = new HashMap<>();
            for (String term : terms) {
                termFreq.merge(term, 1, Integer::sum);
            }
            
            for (Map.Entry<String, Integer> entry : termFreq.entrySet()) {
                invertedIndex.computeIfAbsent(entry.getKey(), 
                    k -> new HashMap<>())
                    .put(docId, entry.getValue());
            }
        }
        
        avgDocumentLength = (double) totalLength / documents.size();
    }
    
    /**
     * 分词
     */
    private List<String> tokenize(String text) {
        // 简单分词，实际应使用jieba等中文分词库
        return Arrays.asList(text.split("[\\s\\p{Punct}]+"))
            .stream()
            .filter(s -> s.length() > 1)
            .map(String::toLowerCase)
            .collect(Collectors.toList());
    }
    
    /**
     * BM25检索
     */
    public List<BM25Result> search(String query, int topK) {
        List<String> queryTerms = tokenize(query);
        
        // 计算每个文档的BM25得分
        Map<String, Double> scores = new HashMap<>();
        
        for (String term : queryTerms) {
            if (!invertedIndex.containsKey(term)) {
                continue;
            }
            
            // 计算IDF
            double idf = calculateIDF(term);
            
            // 遍历包含该词的文档
            for (Map.Entry<String, Integer> entry : 
                    invertedIndex.get(term).entrySet()) {
                String docId = entry.getKey();
                int tf = entry.getValue();
                
                double docLength = documentLengths.get(docId);
                
                // BM25公式
                double score = idf * (tf * (k1 + 1)) / 
                    (tf + k1 * (1 - b + b * docLength / avgDocumentLength));
                
                scores.merge(docId, score, Double::sum);
            }
        }
        
        // 排序返回topK
        return scores.entrySet().stream()
            .sorted((e1, e2) -> Double.compare(e2.getValue(), e1.getValue()))
            .limit(topK)
            .map(e -> {
                BM25Result result = new BM25Result();
                result.setDocId(e.getKey());
                result.setScore(e.getValue());
                return result;
            })
            .collect(Collectors.toList());
    }
    
    /**
     * 计算IDF
     */
    private double calculateIDF(String term) {
        int docFreq = invertedIndex.containsKey(term) ? 
            invertedIndex.get(term).size() : 0;
        
        int N = documentLengths.size();
        
        // 平滑的IDF公式
        return Math.log((N - docFreq + 0.5) / (docFreq + 0.5) + 1);
    }
}
```

```java
/**
 * 知识图谱检索服务
 */
@Service
public class KnowledgeGraphSearchService {
    
    @Autowired
    private Neo4jTemplate neo4jTemplate;
    
    @Autowired
    private KnowledgeGraphService knowledgeGraph;
    
    /**
     * 基于概念的检索
     */
    public List<KGSearchResult> searchByConcept(String concept) {
        String cypher = """
            MATCH (c:Concept)
            WHERE c.name CONTAINS $concept OR c.description CONTAINS $concept
            OPTIONAL MATCH (c)-[r]-(related)
            RETURN c, collect({rel: r, node: related}) as relations
            """;
        
        Map<String, Object> params = new HashMap<>();
        params.put("concept", concept);
        
        return executeQuery(cypher, params);
    }
    
    /**
     * 基于路径的检索
     * 查找两个概念之间的关系路径
     */
    public List<Path> findRelationPath(String concept1, String concept2) {
        String cypher = """
            MATCH path = shortestPath(
                (c1:Concept {name: $concept1})-[*]-(c2:Concept {name: $concept2})
            )
            RETURN path
            """;
        
        Map<String, Object> params = new HashMap<>();
        params.put("concept1", concept1);
        params.put("concept2", concept2);
        
        return neo4jTemplate.findAll(cypher, params, Path.class);
    }
    
    /**
     * 基于子图的检索
     * 查找与给定概念相关的子图
     */
    public SubGraph searchSubgraph(String concept, int depth) {
        String cypher = """
            MATCH (center:Concept {name: $concept})
            CALL apoc.path.subgraphAll(center, {
                maxLevel: $depth
            })
            YIELD nodes, relationships
            RETURN nodes, relationships
            """;
        
        Map<String, Object> params = new HashMap<>();
        params.put("concept", concept);
        params.put("depth", depth);
        
        return neo4jTemplate.findOne(cypher, params, SubGraph.class);
    }
    
    /**
     * 混合查询
     * 结合多种图查询
     */
    public List<KGSearchResult> hybridSearch(String query, 
                                              List<String> entities) {
        List<KGSearchResult> results = new ArrayList<>();
        
        // 1. 精确匹配
        for (String entity : entities) {
            results.addAll(searchByConcept(entity));
        }
        
        // 2. 相关概念扩展
        Set<String> relatedConcepts = new HashSet<>();
        for (KGSearchResult result : results) {
            relatedConcepts.addAll(
                knowledgeGraph.getRelatedConcepts(result.getConcept().getId())
            );
        }
        
        for (String related : relatedConcepts) {
            if (!query.toLowerCase().contains(related.toLowerCase())) {
                results.addAll(searchByConcept(related));
            }
        }
        
        // 去重和排序
        return results.stream()
            .distinct()
            .sorted((r1, r2) -> 
                Double.compare(r2.getRelevanceScore(), r1.getRelevanceScore()))
            .collect(Collectors.toList());
    }
}
```

### 4.4 多路召回融合

```java
/**
 * 多路召回融合器
 * 使用RRF算法融合多路检索结果
 */
@Service
public class HybridRetrievalService {
    
    private final VectorSearchService vectorSearch;
    private final BM25SearchService bm25Search;
    private final KnowledgeGraphSearchService kgSearch;
    
    /**
     * 混合检索
     */
    public List<FusedResult> hybridSearch(
            QuestionAnalysis analysis,
            int topK) {
        
        String query = analysis.getSearchQuery();
        List<String> keywords = analysis.getKeywords();
        
        // 1. 并行执行三路检索
        CompletableFuture<List<SearchResult>> vectorFuture = 
            CompletableFuture.supplyAsync(() -> 
                vectorSearch.semanticSearch(query, topK * 2));
        
        CompletableFuture<List<SearchResult>> bm25Future = 
            CompletableFuture.supplyAsync(() -> 
                bm25Search.search(query, topK * 2));
        
        CompletableFuture<List<SearchResult>> kgFuture = 
            CompletableFuture.supplyAsync(() -> 
                kgSearch.hybridSearch(query, keywords).stream()
                    .map(this::toSearchResult)
                    .collect(Collectors.toList()));
        
        // 等待所有检索完成
        CompletableFuture.allOf(vectorFuture, bm25Future, kgFuture).join();
        
        // 2. 获取结果
        List<SearchResult> vectorResults = vectorFuture.get();
        List<SearchResult> bm25Results = bm25Future.get();
        List<SearchResult> kgResults = kgFuture.get();
        
        // 3. RRF融合
        return reciprocalRankFusion(
            vectorResults, bm25Results, kgResults, topK
        );
    }
    
    /**
     * 倒数排序融合算法 (RRF)
     */
    private List<FusedResult> reciprocalRankFusion(
            List<SearchResult>... resultLists) {
        
        // 权重配置
        double[] weights = {0.3, 0.3, 0.4};  // vector, bm25, kg
        
        // 累积得分
        Map<String, Double> scores = new HashMap<>();
        Map<String, SearchResult> resultMap = new HashMap<>();
        
        for (int listIdx = 0; listIdx < resultLists.length; listIdx++) {
            List<SearchResult> results = resultLists[listIdx];
            double weight = weights[listIdx];
            
            for (int rank = 0; rank < results.size(); rank++) {
                SearchResult result = results.get(rank);
                String docId = result.getDocId();
                
                // RRF得分 = 1 / (k + rank)
                double rrfScore = weight * (1.0 / (60 + rank + 1));
                
                scores.merge(docId, rrfScore, Double::sum);
                resultMap.put(docId, result);
            }
        }
        
        // 排序并返回
        return scores.entrySet().stream()
            .sorted((e1, e2) -> 
                Double.compare(e2.getValue(), e1.getValue()))
            .map(entry -> {
                FusedResult fused = new FusedResult();
                fused.setDocId(entry.getKey());
                fused.setScore(entry.getValue());
                fused.setResult(resultMap.get(entry.getKey()));
                return fused;
            })
            .collect(Collectors.toList());
    }
    
    /**
     * 重排序
     * 使用更精确的模型对融合结果进行重排
     */
    public List<FusedResult> rerank(
            List<FusedResult> results,
            String query) {
        
        // 使用BERT等交叉编码器进行精排
        for (FusedResult result : results) {
            double rerankScore = crossEncoderScore(query, 
                result.getResult().getContent());
            result.setScore(result.getScore() * 0.5 + rerankScore * 0.5);
        }
        
        // 重新排序
        return results.stream()
            .sorted((r1, r2) -> 
                Double.compare(r2.getScore(), r1.getScore()))
            .collect(Collectors.toList());
    }
}
```

```java
public class RAGGenerationService {
    
    private final OpenAIApi openAIApi;
    private final PromptTemplate promptTemplate;
    private final ContentFormatter formatter;
    
    /**
     * 生成答案
     */
    public GeneratedAnswer generate(String question,
                                    List<RetrievedContext> contexts,
                                    ConversationHistory history) {
        
        // 1. 组装上下文
        String context = assembleContext(contexts);
        
        // 2. 构建提示词
        String prompt = buildPrompt(question, context, history);
        
        // 3. 调用大模型生成
        ChatMessage response = callLLM(prompt);
        
        // 4. 提取引用
        List<Citation> citations = extractCitations(response, contexts);
        
        // 5. 构建答案
        GeneratedAnswer answer = new GeneratedAnswer();
        answer.setContent(response.getContent());
        answer.setCitations(citations);
        answer.setConfidence(calculateConfidence(response));
        answer.setModelUsed(response.getModel());
        
        return answer;
    }
    
    /**
     * 组装检索上下文
     */
    private String assembleContext(List<RetrievedContext> contexts) {
        if (contexts.isEmpty()) {
            return "没有找到相关信息。";
        }
        
        StringBuilder sb = new StringBuilder();
        sb.append("【参考信息】\n\n");
        
        for (int i = 0; i < contexts.size(); i++) {
            RetrievedContext ctx = contexts.get(i);
            sb.append("[").append(i + 1).append("] ");
            sb.append(ctx.getTitle()).append("\n");
            sb.append(ctx.getContent()).append("\n\n");
        }
        
        return sb.toString();
    }
    
    /**
     * 构建提示词
     */
    private String buildPrompt(String question,
                               String context,
                               ConversationHistory history) {
        String systemPrompt = """
            你是一个专业的教育助手。请根据提供的参考信息回答用户的问题。
            
            回答要求：
            1. 仅根据参考信息回答，不要编造信息
            2. 如果信息不足以回答，请明确说明
            3. 回答要准确、清晰、有条理
            4. 适当引用参考信息的编号
            5. 对于复杂概念，使用通俗易懂的语言解释
            """;
        
        String historyContext = "";
        if (history != null && !history.getMessages().isEmpty()) {
            historyContext = "\n【对话历史】\n" + 
                history.formatRecentMessages(3) + "\n";
        }
        
        return String.format("""
            %s
            
            %s
            
            【当前问题】
            %s
            
            %s
            """, 
            systemPrompt,
            context,
            question,
            historyContext
        );
    }
    
    /**
     * 调用大模型
     */
    private ChatMessage callLLM(String prompt) {
        try {
            ChatCompletionRequest request = ChatCompletionRequest.builder()
                .model("glm-4")
                .messages(Arrays.asList(
                    ChatMessage.builder()
                        .role("user")
                        .content(prompt)
                        .build()
                ))
                .temperature(0.7)
                .maxTokens(2000)
                .build();
            
            ChatCompletionResult result = openAIApi.createChatCompletion(request);
            
            return ChatMessage.builder()
                .content(result.getChoices().get(0).getMessage().getContent())
                .model(result.getModel())
                .build();
                
        } catch (Exception e) {
            throw new RuntimeException("LLM调用失败", e);
        }
    }
    
    /**
     * 提取引用
     */
    private List<Citation> extractCitations(ChatMessage response,
                                           List<RetrievedContext> contexts) {
        List<Citation> citations = new ArrayList<>();
        
        String content = response.getContent();
        
        // 查找[n]格式的引用
        Pattern pattern = Pattern.compile("\\[(\\d+)\\]");
        Matcher matcher = pattern.matcher(content);
        
        Set<Integer> citedIndices = new HashSet<>();
        while (matcher.find()) {
            int index = Integer.parseInt(matcher.group(1)) - 1;
            if (index >= 0 && index < contexts.size()) {
                citedIndices.add(index);
            }
        }
        
        for (int index : citedIndices) {
            RetrievedContext ctx = contexts.get(index);
            Citation citation = new Citation();
            citation.setIndex(index + 1);
            citation.setDocId(ctx.getDocId());
            citation.setTitle(ctx.getTitle());
            citation.setRelevance(ctx.getScore());
            
            citations.add(citation);
        }
        
        return citations;
    }
    
    /**
     * 计算置信度
     */
    private double calculateConfidence(ChatMessage response) {
        // 基于多个因素评估置信度
        double confidence = 0.8;  // 基础置信度
        
        String content = response.getContent();
        
        // 降低置信度的情况
        if (content.contains("无法") || content.contains("不足以")) {
            confidence -= 0.2;
        }
        if (content.contains("不确定") || content.contains("可能")) {
            confidence -= 0.1;
        }
        
        return Math.max(0.0, Math.min(1.0, confidence));
    }
}
```

```java
/**

@Service
public class ConversationContextManager {
    
    private final Map<String, ConversationContext> contexts;
    private final int MAX_HISTORY_SIZE = 10;
    private final long CONTEXT_EXPIRE_HOURS = 24;
    
    /**
     * 获取或创建对话上下文
     */
    public ConversationContext getContext(String sessionId) {
        ConversationContext context = contexts.get(sessionId);
        
        if (context == null) {
            context = new ConversationContext();
            context.setSessionId(sessionId);
            context.setMessages(new ArrayList<>());
            context.setCreatedAt(System.currentTimeMillis());
            contexts.put(sessionId, context);
        } else {
            // 检查是否过期
            long expireTime = context.getCreatedAt() + 
                CONTEXT_EXPIRE_HOURS * 60 * 60 * 1000;
            if (System.currentTimeMillis() > expireTime) {
                context = new ConversationContext();
                context.setSessionId(sessionId);
                context.setMessages(new ArrayList<>());
                context.setCreatedAt(System.currentTimeMillis());
                contexts.put(sessionId, context);
            }
        }
        
        return context;
    }
    
    /**
     * 添加消息
     */
    public void addMessage(String sessionId, Message message) {
        ConversationContext context = getContext(sessionId);
        context.getMessages().add(message);
        
        // 保持历史消息数量限制
        if (context.getMessages().size() > MAX_HISTORY_SIZE) {
            context.getMessages().remove(0);
        }
        
        // 更新最后活跃时间
        context.setLastActiveAt(System.currentTimeMillis());
    }
    
    /**
     * 获取相关上下文
     */
    public String getRelevantContext(String sessionId, String currentQuery) {
        ConversationContext context = getContext(sessionId);
        List<Message> messages = context.getMessages();
        
        if (messages.isEmpty()) {
            return "";
        }
        
        // 选择与当前查询相关的历史消息
        List<Message> relevantMessages = new ArrayList<>();
        
        for (Message msg : messages) {
            double similarity = calculateSimilarity(
                currentQuery, 
                msg.getContent()
            );
            
            if (similarity > 0.3) {
                relevantMessages.add(msg);
            }
        }
        
        // 如果没有相关的，返回最近的消息
        if (relevantMessages.isEmpty()) {
            int recentCount = Math.min(3, messages.size());
            relevantMessages = messages.subList(
                messages.size() - recentCount, 
                messages.size()
            );
        }
        
        return formatContextMessages(relevantMessages);
    }
    
    /**
     * 格式化为上下文字符串
     */
    private String formatContextMessages(List<Message> messages) {
        StringBuilder sb = new StringBuilder();
        sb.append("【之前的对话】\n");
        
        for (Message msg : messages) {
            sb.append(msg.getRole()).append(": ");
            sb.append(msg.getContent()).append("\n");
        }
        
        return sb.toString();
    }
    
    /**
     * 计算相似度
     */
    private double calculateSimilarity(String text1, String text2) {
        // 使用简单的词重叠计算
        Set<String> words1 = new HashSet<>(Arrays.asList(
            text1.toLowerCase().split("\\s+")
        ));
        Set<String> words2 = new HashSet<>(Arrays.asList(
            text2.toLowerCase().split("\\s+")
        ));
        
        Set<String> intersection = new HashSet<>(words1);
        intersection.retainAll(words2);
        
        Set<String> union = new HashSet<>(words1);
        union.addAll(words2);
        
        if (union.isEmpty()) {
            return 0;
        }
        
        return (double) intersection.size() / union.size();
    }
}
```

```java
/**
 * 答案质量评估服务
 */
@Service
public class AnswerQualityEvaluator {
    
    /**
     * 评估答案质量
     */
    public QualityReport evaluate(GeneratedAnswer answer,
                                  String question,
                                  List<RetrievedContext> contexts) {
        QualityReport report = new QualityReport();
        
        // 1. 准确性评估
        report.setAccuracyScore(evaluateAccuracy(answer, contexts));
        
        // 2. 完整性评估
        report.setCompletenessScore(evaluateCompleteness(
            answer, question, contexts));
        
        // 3. 连贯性评估
        report.setCoherenceScore(evaluateCoherence(answer));
        
        // 4. 引用准确性
        report.setCitationScore(evaluateCitations(answer, contexts));
        
        // 5. 综合评分
        double overallScore = 
            report.getAccuracyScore() * 0.3 +
            report.getCompletenessScore() * 0.25 +
            report.getCoherenceScore() * 0.2 +
            report.getCitationScore() * 0.25;
        report.setOverallScore(overallScore);
        
        // 6. 质量等级
        report.setGrade(determineGrade(overallScore));
        
        // 7. 问题检测
        report.setIssues(detectIssues(report));
        
        return report;
    }
    
    /**
     * 评估准确性
     */
    private double evaluateAccuracy(GeneratedAnswer answer,
                                    List<RetrievedContext> contexts) {
        String content = answer.getContent();
        
        // 检查是否包含"不知道"等不确定表达
        if (content.contains("不知道") || 
            content.contains("无法确定") ||
            content.contains("信息不足")) {
            return 0.9;  // 正确表达了不确定性
        }
        
        // 检查是否与参考内容一致
        int matchCount = 0;
        for (RetrievedContext ctx : contexts) {
            if (containsClaimedInfo(content, ctx.getContent())) {
                matchCount++;
            }
        }
        
        return Math.min(1.0, (double) matchCount / contexts.size());
    }
    
    /**
     * 评估完整性
     */
    private double evaluateCompleteness(String answer,
                                       String question,
                                       List<RetrievedContext> contexts) {
        double score = 0.5;  // 基础分
        
        // 检查是否回答了问题的各个方面
        if (question.contains("为什么")) {
            if (answer.contains("因为") || answer.contains("原因")) {
                score += 0.25;
            }
        }
        
        if (question.contains("如何") || question.contains("怎么")) {
            if (answer.contains("步骤") || answer.contains("首先") ||
                answer.contains("然后")) {
                score += 0.25;
            }
        }
        
        if (question.contains("区别")) {
            if (answer.contains("而") || answer.contains("但是") ||
                answer.contains("不同")) {
                score += 0.25;
            }
        }
        
        // 检查是否有足够的详细信息
        if (answer.length() > 100) {
            score += 0.25;
        }
        
        return Math.min(1.0, score);
    }
    
    /**
     * 评估连贯性
     */
    private double evaluateCoherence(GeneratedAnswer answer) {
        String content = answer.getContent();
        double score = 1.0;
        
        // 检查逻辑连接词
        if (!content.contains("因为") && !content.contains("所以") &&
            !content.contains("但是") && !content.contains("此外")) {
            score -= 0.1;
        }
        
        // 检查句子完整性
        String[] sentences = content.split("[。！？]");
        int incompleteCount = 0;
        for (String sentence : sentences) {
            if (sentence.trim().length() < 5) {
                incompleteCount++;
            }
        }
        
        if (incompleteCount > sentences.length * 0.3) {
            score -= 0.2;
        }
        
        return Math.max(0.0, score);
    }
    
    /**
     * 评估引用
     */
    private double evaluateCitations(GeneratedAnswer answer,
                                     List<RetrievedContext> contexts) {
        List<Citation> citations = answer.getCitations();
        
        if (citations.isEmpty()) {
            return 0.5;  // 没有引用给基础分
        }
        
        double score = 0.5;
        
        // 检查引用是否有效
        for (Citation citation : citations) {
            if (citation.getIndex() > 0 && 
                citation.getIndex() <= contexts.size()) {
                score += 0.1;
            }
        }
        
        return Math.min(1.0, score);
    }
    
    /**
     * 确定质量等级
     */
    private QualityGrade determineGrade(double score) {
        if (score >= 0.9) return QualityGrade.EXCELLENT;
        if (score >= 0.8) return QualityGrade.GOOD;
        if (score >= 0.6) return QualityGrade.ACCEPTABLE;
        if (score >= 0.4) return QualityGrade.POOR;
        return QualityGrade.FAIL;
    }
}
```

```java
/**
 * 问答控制器
 */
@RestController
@RequestMapping("/api/qa")
public class QAController {
    
    @Autowired
    private QuestionUnderstandingService questionUnderstanding;
    
    @Autowired
    private HybridRetrievalService hybridRetrieval;
    
    @Autowired
    private RAGGenerationService ragGeneration;
    
    @Autowired
    private AnswerQualityEvaluator qualityEvaluator;
    
    @Autowired
    private ConversationContextManager contextManager;
    
    /**
     * 问答接口
     */
    @PostMapping("/ask")
    public ResponseEntity<QAResponse> ask(@RequestBody QARequest request) {
        String question = request.getQuestion();
        String sessionId = request.getSessionId();
        
        // 1. 问题理解
        QuestionAnalysis analysis = questionUnderstanding.analyze(question);
        
        // 2. 混合检索
        List<FusedResult> retrievalResults = hybridRetrieval.hybridSearch(
            analysis, 10
        );
        
        // 3. 重排序
        retrievalResults = hybridRetrieval.rerank(
            retrievalResults, question
        );
        
        // 4. 取Top结果
        List<RetrievedContext> topContexts = retrievalResults.stream()
            .limit(5)
            .map(FusedResult::getResult)
            .map(this::toRetrievedContext)
            .collect(Collectors.toList());
        
        // 5. 获取对话上下文
        String contextStr = contextManager.getRelevantContext(
            sessionId, question
        );
        ConversationHistory history = parseHistory(contextStr);
        
        // 6. 生成答案
        GeneratedAnswer answer = ragGeneration.generate(
            question, topContexts, history
        );
        
        // 7. 质量评估
        QualityReport quality = qualityEvaluator.evaluate(
            answer, question, topContexts
        );
        
        // 8. 保存对话历史
        contextManager.addMessage(sessionId, 
            Message.userMessage(question));
        contextManager.addMessage(sessionId, 
            Message.assistantMessage(answer.getContent()));
        
        // 9. 构建响应
        QAResponse response = new QAResponse();
        response.setAnswer(answer.getContent());
        response.setCitations(answer.getCitations());
        response.setConfidence(answer.getConfidence());
        response.setQuality(quality);
        response.setSessionId(sessionId);
        
        return ResponseEntity.ok(response);
    }
    
    /**
     * 反馈接口
     */
    @PostMapping("/feedback")
    public ResponseEntity<Void> submitFeedback(
            @RequestBody FeedbackRequest request) {
        
        feedbackCollector.collectFeedback(
            request.getAnswerId(),
            request.getUserId(),
            request.getType(),
            request.getFeedback()
        );
        
        return ResponseEntity.ok().build();
    }
}
```
